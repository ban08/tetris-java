package controller.command;

import controller.ControllerInterface;

/**
 * A command that rotates the current piece right.
 */
public final class RotateRightCommand implements Command {

    /**
     * The controller to use for rotating the current piece right.
     */
    private final ControllerInterface controller;

    /**
     * Creates a new RotateRightCommand with the given controller.
     *
     * @param controller the controller to use for rotating the current piece right
     */
    public RotateRightCommand(ControllerInterface controller) {
        this.controller = controller;
    }

    /**
     * Rotates the current piece right.
     */
    @Override
    public void execute() {
        controller.rotateRight();
    }
}
